pipeline {
    agent any

    tools {
        maven 'Maven'
        jdk 'jdk-11'
    }

    stages {

        stage('Clean Workspace') {
            steps {
                deleteDir()
            }
        }

        stage('Clone Backend Repo') {
            steps {
                dir('backend') {
                    git branch: 'main',
                        url: 'https://github.com/kirar1234/ORS-PROJECT-10-BACKEND.git'
                }
            }
        }

        stage('Clone Frontend Repo') {
            steps {
                dir('frontend') {
                    git branch: 'main',
                        url: 'https://github.com/kirar1234/orsproject-10-main.git'
                }
            }
        }

        stage('Build Angular') {
            steps {
                dir('frontend') {
                    bat 'npm install'
                    bat 'npm run build'
                }
            }
        }

        stage('Build Spring Boot') {
            steps {
                dir('backend/ORSProject-10(MAIN)') {
                    bat 'mvn clean package -DskipTests'
                }
            }
        }

        stage('Stop Old Containers') {
            steps {
                bat 'docker-compose -f backend/ORSProject-10(MAIN)/docker-compose.yml down'
            }
        }

        stage('Build Docker Images') {
            steps {
                bat 'docker-compose -f backend/ORSProject-10(MAIN)/docker-compose.yml build'
            }
        }

        stage('Run Containers') {
            steps {
                bat 'docker-compose -f backend/ORSProject-10(MAIN)/docker-compose.yml up -d'
            }
        }

        stage('Check Containers') {
            steps {
                bat 'docker ps'
            }
        }

    }
}
